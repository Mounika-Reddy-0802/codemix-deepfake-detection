# Pinned dependencies for the code-mixed audio deepfake detection project.
# Python 3.10+.  Versions pinned for reproducibility (verify availability on
# your Kaggle/Colab base image; some Kaggle images pre-pin torch).
#
# NOTE: CI installs only a light subset (ruff, pytest, numpy, pandas, pyyaml)
# so linting/tests stay fast — the heavy ML/telephony wheels below are for the
# training and live-call environments, not the CI runner.

# --- Core DL / audio ---
# Lower bound: 2.5.1 has no Python 3.13 wheel; 2.6 is the first that does.
# Upper bound: from 2.9 torchaudio routes all audio IO through `torchcodec`, which
# loads FFmpeg's shared libraries and fails against FFmpeg 8/9 (`WinError 127` on
# libtorchcodec_image.dll). 2.8 uses torchaudio's own backends and works.
# Keep torch/torchaudio/torchvision on the SAME minor -- a mismatched torchvision
# raises "operator torchvision::nms does not exist" from inside transformers.
torch>=2.6,<2.9
torchaudio>=2.6,<2.9
torchvision>=0.21,<0.24   # not used directly; transformers imports it
# 4.57 is the floor coqui-tts requires; 5.x dropped transformers.pytorch_utils.
# isin_mps_friendly, which XTTS-v2 imports -- installing it breaks generation with
# an ImportError at `from TTS.api import TTS`. Keep the upper bound.
transformers>=4.57,<5
peft==0.13.2
librosa==0.10.2.post1
soundfile==0.12.1
silero-vad==5.1.2

# --- Voice cloning (spoof generation only; not needed to train or serve) ---
# The original `TTS` package stops at Python 3.11. `coqui-tts` is the maintained
# fork and packages the same XTTS-v2 weights for current Python -- it is what runs
# on this project's Python 3.13 laptop and on Colab.
coqui-tts==0.27.5
# From torch 2.9 the TTS audio IO path requires torchcodec; without it `from
# TTS.api import TTS` raises. Harmless on older torch.
torchcodec==0.16.0
